// forge:tokens ApplicationAuthNamespace, ApplicationPersistenceNamespace, InfrastructureNamespace,
// forge:tokens DbContextNamespace, DbContextName, ApiMiddlewareNamespace, Solution,
// forge:tokens SwaggerRegistration, SwaggerMiddleware
{{ForgeRuntimeUsing}}using Microsoft.EntityFrameworkCore;
using Wolverine;
using {{ApiMiddlewareNamespace}};
using {{ApplicationAuthNamespace}};
using {{ApplicationPersistenceNamespace}};
using {{DbContextNamespace}};
using {{InfrastructureNamespace}};

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();
{{SwaggerRegistration}}
// SQLite by default: no server to install, and unlike InMemory it is a relational provider, so
// migrations work out of the box. Swap for UseSqlServer/UseNpgsql when you have one.
builder.Services.AddDbContext<{{DbContextName}}>(options =>
    options.UseSqlite(builder.Configuration.GetConnectionString("Default")
                      ?? "Data Source={{Solution}}.db"));

{{CurrentUserRegistration}}

// Repositories and IUnitOfWork. make:repo appends new registrations to AddInfrastructure.
builder.Services.AddInfrastructure();
{{ForgeWolverineRegistration}}

builder.Host.UseWolverine(options =>
{
    // Handlers live in the ApplicationService project, but Wolverine only scans the application
    // assembly by default — so without this, every handler make:feature generates is silently
    // never discovered and the message has no handler at runtime.
    options.Discovery.IncludeAssembly(typeof(IUnitOfWork).Assembly);

    // Secure by default: every message is denied unless it opts in. See RoleCheckMiddleware.
    options.Policies.AddMiddleware(typeof(RoleCheckMiddleware));
});

var app = builder.Build();
{{ForgeRuntimeHook}}{{SwaggerMiddleware}}
app.UseHttpsRedirection();
app.MapControllers();

app.Run();
